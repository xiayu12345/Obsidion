+static snd_pcm_sframes_t writePcmWithRecover(SoundCtrlContext* sc, snd_pcm_t *pcm,
+        void *pData, snd_pcm_uframes_t num_frames)
+{
+    snd_pcm_uframes_t written = 0;
+    snd_pcm_uframes_t remaining = num_frames;
+    char *ptr = (char*)pData;
+    size_t frame_bytes = masterFrameBytes(sc);
+
+    if (!pcm)
+        return -EINVAL;
+    if (num_frames == 0)
+        return 0;
+    if (frame_bytes == 0)
+        frame_bytes = 4;
+
+    while (remaining > 0) {
+        snd_pcm_sframes_t res = snd_pcm_writei(pcm, ptr, remaining);
+
+        if (res == -EINTR) {
+            continue;
+        }

+        }
